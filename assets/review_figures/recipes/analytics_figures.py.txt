def roc(y,s):
 y=np.asarray(y);s=np.asarray(s);assert set(np.unique(y))=={0,1};assert np.isfinite(s).all()
 order=np.argsort(-s,kind='stable');yy=y[order];ss=s[order];ends=np.r_[np.where(np.diff(ss)!=0)[0],len(ss)-1]
 tp=np.r_[0,np.cumsum(yy)[ends]]/yy.sum();fp=np.r_[0,np.cumsum(1-yy)[ends]]/(len(yy)-yy.sum())
 auc=float(np.trapezoid(tp,fp));return fp,tp,auc

# Tie-aware complete precision-recall curves from fixed outer-fold scores.
def precision_recall(y,score):
 y=np.asarray(y,dtype=int);score=np.asarray(score);assert set(np.unique(y))=={0,1};assert np.isfinite(score).all()
 order=np.argsort(-score,kind='stable');yy=y[order];vv=score[order]
 ends=np.r_[np.flatnonzero(np.diff(vv)),len(vv)-1];tp=np.cumsum(yy)[ends]
 recall=tp/yy.sum();precision=tp/(ends+1)
 ap=float(np.sum(np.diff(np.r_[0,recall])*precision))
 return np.r_[0,recall],np.r_[1,precision],ap
