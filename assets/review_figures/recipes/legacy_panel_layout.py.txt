"""Register original PDF plot areas against native axes; never estimate new data.

Source paths (including error bars) remain PDF vectors. Only labels are reset.
Coordinates below are source-page points, not reconstructed observations.
"""
import numpy as np
import pymupdf as fitz
from matplotlib.patches import Patch
from matplotlib.lines import Line2D

LAYERS=[]
AUDIT=[]

def prepare(ax, source, frame, xlabels=None, ylabels=None, erase=()):
    doc=fitz.open(source);page=doc[0];paths=page.get_drawings()
    # Snap the reviewed frame coordinates to the original, full-precision spines.
    r=fitz.Rect(frame)
    vertical=[p['rect'] for p in paths if p['rect'].width<.01 and p['rect'].height>r.height*.9]
    horizontal=[p['rect'] for p in paths if p['rect'].height<.01 and p['rect'].width>r.width*.9]
    coords=[min(vertical,key=lambda p:abs(p.x0-r.x0)).x0,
            min(horizontal,key=lambda p:abs(p.y0-r.y0)).y0,
            min(vertical,key=lambda p:abs(p.x0-r.x1)).x0,
            min(horizontal,key=lambda p:abs(p.y0-r.y1)).y0]
    assert max(abs(a-b) for a,b in zip(coords,frame))<.06,(source,frame,coords)
    r=fitz.Rect(coords)
    xt=sorted({p['rect'].x0 for p in paths if p['rect'].width<.01 and 1<p['rect'].height<3 and abs(p['rect'].y0-r.y1)<.02 and r.x0-.02<=p['rect'].x0<=r.x1+.02})
    yt=sorted({p['rect'].y0 for p in paths if p['rect'].height<.01 and 1<p['rect'].width<3 and abs(p['rect'].x1-r.x0)<.02 and r.y0-.02<=p['rect'].y0<=r.y1+.02},reverse=True)
    ax.grid(False);ax.set(xlim=(r.x0,r.x1),ylim=(r.y1,r.y0))
    if xlabels is None:ax.set_xticks([])
    else:
        assert len(xt)==len(xlabels),(source,'x',xt,xlabels)
        ax.set_xticks(xt,xlabels)
    if ylabels is None:ax.set_yticks([])
    else:
        assert len(yt)==len(ylabels),(source,'y',yt,ylabels)
        ax.set_yticks(yt,ylabels)
    # A text-only redaction leaves every data path untouched.
    for block in page.get_text('dict')['blocks']:
        for line in block.get('lines',[]):page.add_redact_annot(fitz.Rect(line['bbox']),fill=False)
    page.apply_redactions(images=0,graphics=0)
    for rect in erase:page.draw_rect(fitz.Rect(rect),color=None,fill=(1,1,1),overlay=True)
    LAYERS.append((ax,doc,r,False))
    pos=ax.get_position();w,h=ax.figure.get_size_inches()*72
    AUDIT.append({'source':str(source),'source_plot_rect':list(r),
                  'target_plot_rect_points':[pos.x0*w,pos.y0*h,pos.x1*w,pos.y1*h],
                  'tick_positions':{'x':xt,'y':yt},'data_paths':'original PDF vectors; no re-estimation'})
    ax._audit_index=len(AUDIT)-1
    return r

def note(ax,x,y,text,**kwargs):
    return ax.text(x,y,text,transform=ax.transAxes,fontsize=6.5,va='top',**kwargs)

def legend(ax,labels,colors,**kwargs):
    ax.legend([Patch(facecolor=c,edgecolor='#83929a',linewidth=.4) for c in colors],labels,
              frameon=False,fontsize=6,handlelength=1.1,labelspacing=.25,**kwargs)

def native_forest(ax, expected):
    # Keep exact PDF center/interval coordinates, with circular point markers in points.
    target,doc,r,_=LAYERS.pop();assert target is ax
    points=[];intervals=[]
    for p in doc[0].get_drawings():
        b=p['rect'];items=p['items']
        if not (r+(-.01,-.01,.01,.01)).contains(b):continue
        if len(items)==8 and all(item[0]=='c' for item in items):
            x=(b.x0+b.x1)/2;y=(b.y0+b.y1)/2
            ax.scatter(x,y,s=27,color=p['fill'],zorder=3);points.append([x,y])
        elif len(items)==1 and items[0][0]=='l':
            if b.height<.01 and r.y0+.05<b.y0<r.y1-.05:
                if b.width>r.width*.95:ax.axhline(b.y0,c='#dfe9ed',lw=.7)
                elif b.width>.1:
                    ax.plot([b.x0,b.x1],[b.y0,b.y0],c=p['color'],lw=1.7)
                    ax.plot([b.x0,b.x1],[b.y0,b.y0],ls='none',marker='|',ms=4,c=p['color'])
                    intervals.append([b.x0,b.y0,b.x1])
            elif b.height>r.height*.9 and r.x0+.05<b.x0<r.x1-.05:
                ax.axvline(b.x0,c='#87949c',lw=.8,ls=(0,(3,3)))
    assert len(points)==expected and len(intervals)==expected,(points,intervals)
    AUDIT[-1]['native_marker_centers_pdf_points']=points
    AUDIT[-1]['native_intervals_pdf_points']=intervals
    doc.close()

def composite(figure, foreground, target):
    out=fitz.open();page=out.new_page(width=foreground[0].rect.width,height=foreground[0].rect.height)
    for ax,doc,source,preserve_aspect in LAYERS:
        pos=ax.get_position();w,h=page.rect.width,page.rect.height
        dest=fitz.Rect(pos.x0*w,(1-pos.y1)*h,pos.x1*w,(1-pos.y0)*h)
        page.show_pdf_page(dest,doc,0,clip=source,keep_proportion=preserve_aspect)
        doc.close()
    page.show_pdf_page(page.rect,foreground,0)
    out.save(target,garbage=4,deflate=True);out.close();LAYERS.clear()
