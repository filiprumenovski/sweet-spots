def roc(y,s):
 y=np.asarray(y);s=np.asarray(s);assert set(np.unique(y))=={0,1};assert np.isfinite(s).all()
 order=np.argsort(-s,kind='stable');yy=y[order];ss=s[order];ends=np.r_[np.where(np.diff(ss)!=0)[0],len(ss)-1]
 tp=np.r_[0,np.cumsum(yy)[ends]]/yy.sum();fp=np.r_[0,np.cumsum(1-yy)[ends]]/(len(yy)-yy.sum())
 auc=float(np.trapezoid(tp,fp));return fp,tp,auc
